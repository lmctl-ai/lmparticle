cmake_minimum_required(VERSION 3.24)
project(lmparticle VERSION 0.2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
option(LMPARTICLE_FULL_STATIC "Link the complete executable statically" ON)

# Embed the schema, the notices and the XCOM tables as byte arrays.
function(embed_file var file ns name)
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${file}")
  file(READ "${file}" hex HEX)
  string(REGEX REPLACE "([0-9a-f][0-9a-f])" "0x\\1," hex "${hex}")
  set(${var} "inline constexpr unsigned char ${name}[] = {${hex}0}; inline constexpr std::size_t ${name}_size = sizeof(${name}) - 1;\n" PARENT_SCOPE)
endfunction()
set(GEN "${CMAKE_CURRENT_BINARY_DIR}/generated")
file(MAKE_DIRECTORY "${GEN}")
embed_file(SCHEMA_SRC "${CMAKE_CURRENT_SOURCE_DIR}/schema/lmparticle.dsl.json" lmparticle embedded_schema)
file(WRITE "${GEN}/schema_embedded.hpp" "#pragma once\n#include <cstddef>\nnamespace lmparticle { ${SCHEMA_SRC} }\n")
embed_file(NOTICES_SRC "${CMAKE_CURRENT_SOURCE_DIR}/THIRD_PARTY_NOTICES" lmparticle embedded_notices)
file(WRITE "${GEN}/notices_embedded.hpp" "#pragma once\n#include <cstddef>\nnamespace lmparticle { ${NOTICES_SRC} }\n")
embed_file(XCOM_SRC "${CMAKE_CURRENT_SOURCE_DIR}/data/xcom.csv" lmparticle embedded_xcom)
file(WRITE "${GEN}/xcom_embedded.hpp" "#pragma once\n#include <cstddef>\nnamespace lmparticle { ${XCOM_SRC} }\n")

add_library(particle_core STATIC cpp/xs.cpp cpp/transport.cpp)
target_include_directories(particle_core PUBLIC cpp "${GEN}")
target_compile_options(particle_core PRIVATE -Wall -Wextra -Wpedantic)
add_executable(lmparticle cpp/main.cpp cpp/memory_budget.cpp)
target_compile_definitions(lmparticle PRIVATE LMPARTICLE_VERSION="${PROJECT_VERSION}")
target_include_directories(lmparticle PRIVATE cpp "${GEN}")
target_link_libraries(lmparticle PRIVATE particle_core)
target_compile_options(lmparticle PRIVATE -Wall -Wextra -Wpedantic)
function(static_executable target)
  if((CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang") AND NOT APPLE)
    target_link_options(${target} PRIVATE -static-libstdc++ -static-libgcc)
    if(LMPARTICLE_FULL_STATIC)
      target_link_options(${target} PRIVATE -static)
    endif()
    if(CMAKE_BUILD_TYPE STREQUAL "Release")
      target_link_options(${target} PRIVATE -s)
    endif()
  endif()
endfunction()
static_executable(lmparticle)

# Native regression tests of the physics core (cross sections, sampling, conservation) without lmscript.
include(CTest)
if(BUILD_TESTING AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/core.cpp")
  add_executable(core_tests tests/core.cpp)
  target_link_libraries(core_tests PRIVATE particle_core)
  target_compile_options(core_tests PRIVATE -Wall -Wextra -Wpedantic)
  static_executable(core_tests)
  add_test(NAME core COMMAND core_tests)
endif()
